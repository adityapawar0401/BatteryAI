        const menuBtn = document.getElementById('menu-btn');

        // Generate Battery Cells
        const grid = document.getElementById('cell-grid');
        for (let i = 0; i < 48; i++) {
            const cell = document.createElement('div');
            // Make 5 cells "warning" status randomly
            const isWarning = Math.random() > 0.85; 
            const statusColor = isWarning ? 'var(--copper)' : 'var(--volt)';
            const statusBg = isWarning ? 'rgba(184, 92, 56, 0.1)' : 'rgba(204, 255, 0, 0.05)';
            
            cell.className = 'aspect-square rounded-sm border border-white/10 flex items-center justify-center mono text-xs text-zinc-600 cursor-pointer transition-all hover:scale-95';
            cell.style.backgroundColor = statusBg;
            cell.innerHTML = `
                <div class="w-full h-full p-1 flex flex-col justify-between">
                    <div class="w-1 h-1 rounded-full" style="background:${statusColor}; margin-left:auto; box-shadow: 0 0 5px ${statusColor};"></div>
                    <div class="text-[8px] text-center text-zinc-500">${(i+1).toString(16).toUpperCase()}</div>
                </div>
            `;
            grid.appendChild(cell);
        }
